// Global Search Filter function for index.html function applySearchFilter() { const query = document.getElementById('global-search').value.toLowerCase().trim(); const tbody = document.getElementById('county-table-body'); const rows = Array.from(tbody.querySelectorAll('tr')); rows.forEach(row => { if (!query) { row.style.display = ''; return; } // Retrieve search terms stored in the row's data attribute const searchData = row.getAttribute('data-search') || ''; if (searchData.includes(query)) { row.style.display = ''; } else { row.style.display = 'none'; } }); } // When rendering table rows in renderCountyTable(items): items.forEach(c => { const tr = document.createElement('tr'); // Aggregate all candidate names into a single searchable string const candNames = (c.candidates || []).map(cand => cand.n || '').join(' '); const searchableText = `${c.name} ${c.type || ''} ${candNames}`.toLowerCase(); tr.setAttribute('data-search', searchableText); // ... rest of row rendering logic }); // Bind search input listener document.getElementById('global-search').addEventListener('input', applySearchFilter);
